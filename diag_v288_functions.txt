=== PITYDAO ===
async function analyzeFirstPityDaomai(){
 if(!SAVE){daoFirstResult.innerHTML='<div class="warn">세이브를 먼저 불러오세요.</div>';return}
 const base=stateFromSave(SAVE,roleObj(),+vipSel.value,rawCalcZizhi());
 const slots=+QUAL[String(base.zizhi)]?.slots||0;
 if(slots<=0){daoFirstResult.innerHTML='<div class="warn">원시 자맥 탭에서 계산용 칸을 1칸 이상 개방하세요.</div>';return}
 const pityAt=+QUAL[String(base.zizhi)]?.base_pity||0;
 if(!pityAt){daoFirstResult.innerHTML='<div class="warn">현재 개방 칸에는 천장 기준이 없습니다.</div>';return}

 daoFirstResult.innerHTML='<div class="warn">원시 #600 이후 천장 직전 일반맥별 seed → 교체맥 15개 계산 중...</div>';
 await uiYield();

 // v22.16: 도맥 제거 경로를 후보마다 다시 세수하지 않는다.
 // 천장 직전으로 당겨질 수 있는 일반맥의 핵심 값은 그 일반맥이 채택된 직후의 BornTalent seed다.
 // 한 번의 원시 normal stream 스캔으로 #600 이후 후보와 seed를 모은 뒤,
 // 각 seed에서 천장 자맥 1개를 뽑고 그 뒤 Luck/교체 자맥 15개를 바로 계산한다.
 const eng=engineFor(base);
 let seed=+base.seed, rc=+(base.born_talent_rc||0), intl=!!base.international_wash;
 let talLucky=+base.tal_lucky||0, rawNo=0, washNo=0;
 const candidates=[];
 const minRaw=600;
 const maxGuard=200000;
 let guard=maxGuard;

 // 잠금 없는 계산용 원시 흐름. 한 페이지 내 중복만 삼킨다.
 while(talLucky<pityAt && guard-->0){
   washNo++;
   const seen=new Set();
   let accepted=0;
   while(accepted<slots && guard-->0){
     let id;
     [id,seed]=rawNormal(seed,eng,intl?rc:0);
     if(intl)rc++;
     if(!eng.allowed.has(id)||seen.has(id))continue;
     const it=BYID.get(id);
     if(!it)continue;
     // 첫 자맥=천장 분기만 다루므로 자연 자맥이 먼저 나오면 이 도맥표 대상이 아니다.
     if(+it.weight===4){
       daoFirstResult.innerHTML=`<div class="warn">원시 #${rawNo+1}에서 자연 자맥 <b>${nm(id)}</b>이 천장보다 먼저 나옵니다. 첫 자맥=천장 도맥법 분기 대상이 아닙니다.</div>`;
       return;
     }
     seen.add(id); accepted++; rawNo++;
     if(rawNo>=minRaw){
       candidates.push({rawNo,washNo,id,name:nm(id),seedAfter:+seed,rcAfter:+rc});
     }
   }
   talLucky+=slots;
   if(washNo%80===0)await uiYield();
 }
 if(guard<=0){daoFirstResult.innerHTML='<div class="bad">원시 흐름 스캔 한도를 초과했습니다.</div>';return}
 if(!candidates.length){daoFirstResult.innerHTML='<div class="warn">원시 #600 이후 천장 직전 후보 일반맥이 없습니다.</div>';return}

 // 후보 일반맥 직후 seed가 바로 다음 천장 삽입의 Luck seed 시작점이라고 보고,
 // 그 seed에서 허용되는 첫 자맥을 천장 자맥으로, 이후 15개를 교체맥으로 뽑는다.
 function purpleFromSeed(startSeed,count){
   let luck=+startSeed, out=[], spins=0;
   while(out.length<count && spins++<200000){
     let id;[id,luck]=rawPurple(luck,eng);
     if(!eng.allowed.has(id))continue;
     out.push({id,name:nm(id),seedAfter:+luck});
   }
   return {rows:out,lastSeed:+luck};
 }

 const rows=[];
 for(let i=0;i<candidates.length;i++){
   const c=candidates[i];
   const p=purpleFromSeed(c.seedAfter,16); // 1개 천장 + 뒤 15개 교체
   if(!p.rows.length)continue;
   rows.push({
     ...c,
     pity:p.rows[0],
     seq:p.rows.slice(1,16),
     luckStart:+c.seedAfter
   });
   if(i%120===119)await uiYield();
 }

 let h=`<div class="good"><b>천장 도맥 seed 표 · 원시 #600 이후</b><br>
 <span class="small">계산용 열린 칸 <b>${slots}칸</b> · 천장 기준 ${pityAt}. 도맥법으로 중복 일반맥을 제거해 천장 직전 일반맥을 바꾸면, 그 일반맥 직후의 <b>BornTalent seed</b>가 달라지고 그 seed에서 천장 자맥과 이후 교체맥 서열이 달라집니다. 후보별 전체 도맥 경로 재시뮬레이션은 하지 않아 속도를 줄였습니다.</span></div>`;
 h+='<div class="scroll"><table><thead><tr><th>직전 원시 #</th><th>천장 직전 일반 자질</th><th>직전 seed</th><th>천장 자맥</th>';
 for(let i=1;i<=15;i++)h+=`<th>교체 ${i}</th>`;
 h+='</tr></thead><tbody>';
 for(const r of rows){
   h+=`<tr><td class="num">${r.rawNo}</td><td>${r.name}</td><td class="num">${r.luckStart}</td><td class="q4">${r.pity.name}</td>`;
   for(let i=0;i<15;i++)h+=`<td class="q4">${r.seq[i]?.name||'-'}</td>`;
   h+='</tr>';
 }
 h+='</tbody></table></div>';
 h+=`<div class="small">표는 #600부터 기본 천장 직전까지의 일반맥을 후보로 봅니다. 같은 이름의 일반 자질이라도 원시 위치가 다르면 seed가 다를 수 있으므로 원시 #별로 따로 표시합니다.</div>`;
 daoFirstResult.innerHTML=h;
}



=== WASHONCE ===
function washOnce(before){
 const s=dc(before),eng=engineFor(s),q=+s.zizhi,slots=+QUAL[String(q)].slots;
 const kept=s.talents.filter(t=>t.locked).map(dc),locks=kept.length;if(locks>=slots)throw Error("모든 슬롯 잠금");
 const cost=COST[slots][locks];let seed=+s.seed,rc=+(s.born_talent_rc||0),luck=+(s.luck_seed||0),linit=!!s.luck_seed_inited;
 const intl=!!s.international_wash,pityArmed=+s.tal_lucky>=+QUAL[String(q)].base_pity;let pityActive=pityArmed;
 let talents=kept,seen=new Set(kept.map(t=>+t.id)),purplePage=false,pityTriggered=false,draws=0;
 const quals={"1":+(s.quality_counts?.["1"]||0),"2":+(s.quality_counts?.["2"]||0),"3":+(s.quality_counts?.["3"]||0),"4":+(s.quality_counts?.["4"]||0)};
 while(talents.length<slots){
   let c;if(pityActive){if(!linit){luck=seed;linit=true}[c,luck]=rawPurple(luck,eng)}
   else{[c,seed]=rawNormal(seed,eng,intl?rc:0);if(intl)rc++}
   if(++draws>100000)throw Error("wash draw did not converge");
   if(!eng.allowed.has(c)||seen.has(c))continue;
   const w=+BYID.get(c).weight;if(w===4&&purplePage)continue;
   talents.push({id:c,locked:false});seen.add(c);quals[String(w)]++;
   if(w===4){purplePage=true;pityTriggered=pityActive;pityActive=false}
 }
 const washNo=+s.total_wash+1,pills=+s.total_pills+cost,prev=s.last_purple_total_pills;
 const gap=!purplePage?null:pills-(prev||0),wasFirst=purplePage&&+s.gold_count===0;
 Object.assign(s,{seed,luck_seed:luck,luck_seed_inited:linit,talents,tal_lucky:purplePage?0:+s.tal_lucky+slots-locks,
 total_wash:washNo,total_talent_bars:+s.total_talent_bars+slots-locks,total_pills:pills,bag_pills:Math.max(0,+s.bag_pills-cost),
 gold_count:+s.gold_count+(purplePage?1:0),pity_count:+s.pity_count+(pityTriggered?1:0),quality_counts:quals});
 if(intl)s.born_talent_rc=rc;if(purplePage)s.last_purple_total_pills=pills;
 return {state:s,result:{wash_no:washNo,talents:dc(talents),got_gold:purplePage,pills_used:cost,total_pills:pills,was_pity:pityTriggered,
 pity_armed:pityArmed,tal_lucky:s.tal_lucky,purple_gap_pills:gap,is_first_purple:wasFirst,raw_draws:draws}};
}
