       else if(!earlier){can=false;reason="동류 충돌맥 없음"}
       else if(priorPurple){can=false;reason="앞선 동류가 자맥뿐"}
       else can=true;
       out.push({wash_no:totalBars,wash_times:totalWash,talent_id:+it.id,talent_name:it.name_ko||it.name,
         was_pity:!!pityActive,earlier_conflict_name:earlier?(earlier[2].name_ko||earlier[2].name):null,
         earlier_conflict_id:earlier?+earlier[2].id:null,earlier_conflict_wash_no:earlier?earlier[0]:null,
         earlier_conflict_wash_times:earlier?earlier[1]:null,earlier_conflict_gap:earlier?totalBars-earlier[0]:null,
         earlier_conflict_is_purple:priorPurple,can_exchange:can,block_reason:reason});
       purplePage=true;pityTriggered=!!pityActive;pityActive=false;
     }
     last.set(type,[totalBars,totalWash,it]);
   }
   talLucky=purplePage?0:talLucky+slots;
   if(z%150===149){if(onProgress)onProgress(totalWash,out.length);await new Promise(r=>setTimeout(r,0))}
 }
 // 신맥은 '첫 천장 전' 다른 신맥이 실제 원시 자맥에 존재해야 최초 무한교체 가능.
 const firstPityIdx=out.findIndex(x=>x.was_pity),pre=firstPityIdx<0?out:out.slice(0,firstPityIdx);
 for(const x of out){
   if(+BYID.get(x.talent_id)?.type!==248)continue;
   const partners=pre.filter(y=>y!==x && +BYID.get(y.talent_id)?.type===248 && y.talent_id!==x.talent_id);
   if(partners.length){
     partners.sort((a,b)=>Math.abs(a.wash_no-x.wash_no)-Math.abs(b.wash_no-x.wash_no));
     const p=partners[0];
     x.can_exchange=!x.was_pity && (firstPityIdx<0 || out.indexOf(x)<firstPityIdx);
     x.earlier_conflict_name=p.talent_name;
     x.earlier_conflict_id=p.talent_id;
     x.earlier_conflict_wash_no=p.wash_no;
     x.earlier_conflict_wash_times=p.wash_times;
     x.block_reason=null;
   }else{
     x.can_exchange=false;
     x.earlier_conflict_name=null;x.earlier_conflict_id=null;x.earlier_conflict_wash_no=null;x.earlier_conflict_wash_times=null;
     x.block_reason="첫 천장 전 다른 신맥 없음";
   }
 }

 // 절대 규칙: 첫 천장이 한 번이라도 나온 뒤의 모든 원시 자맥은 신규 무한교체 시작 불가.
 let pitySeen=false;
 for(const x of out){
   x.hasPriorPity=pitySeen;
   if(pitySeen){
     x.can_exchange=false;
     x.block_reason="앞에서 천장 발생 → 이후 무한교체 시작 불가";
   }
   if(x.was_pity){
     x.can_exchange=false;
     x.block_reason="천장 자맥 → 무한교체 시작 불가";
     pitySeen=true;
   }
 }
 return out;
}
