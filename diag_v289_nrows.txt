function exchangeNRows(base,c,maxN=30,count=15){
 if(c.canInitialize===false)throw Error("이 자맥은 최초 무한교체를 시작할 수 없음");
 let s=dc(base);
 s.talents=s.talents.map(t=>({...t,locked:false}));
 s.bag_pills=999999999;s.has_bag_pills=true;

 const fixed=!!s.luck_seed_inited;
 if(fixed){
   return [{n:0,pills:0,luck:+s.luck_seed,draws:exchangeSequence(+s.luck_seed,s,count)}];
 }
 if(c.was_pity||c.hasPriorPity)throw Error("천장 또는 천장 이후 자맥은 신규 무한교체 불가");
 if(!c.conflictId)throw Error("가장 가까운 충돌맥이 없음");

 const target=+c.talent_id, conflict=+c.conflictId, conflictWash=+c.conflictWashTimes;
 if(!conflictWash)throw Error("가장 가까운 충돌맥 회차 정보 없음");

 // A. 원시표에 기록된 가장 가까운 충돌맥 회차까지 무잠금으로 정확히 재현.
 while(+s.total_wash<conflictWash)s=washOnce(s).state;
 if(+s.total_wash!==conflictWash)throw Error("충돌맥 회차 재현 실패");

 let ci=s.talents.findIndex(x=>+x.id===conflict);
 if(ci<0)throw Error(`가장 가까운 충돌맥 ${nm(conflict)}이 원시 회차 ${conflictWash} 페이지에 없음`);

 // B. 그 충돌맥만 즉시 잠금.
 s.talents=s.talents.map((x,i)=>({...x,locked:i===ci}));

 // C. 그 잠금을 유지한 채 선택한 자맥이 뜰 때까지 세수하고 자맥도 잠금.
 let ti=s.talents.findIndex(x=>+x.id===target);
 if(ti<0){
   let guard=30000;
   while(guard-->0){
     s=washOnce(s).state;
     ci=s.talents.findIndex(x=>+x.id===conflict);
     if(ci<0)throw Error(`잠근 충돌맥 ${nm(conflict)}이 유지되지 않음`);
     s.talents[ci].locked=true;
     ti=s.talents.findIndex(x=>+x.id===target);
     if(ti>=0)break;
   }
   if(ti<0)throw Error(`목표 자맥 ${nm(target)}을 찾지 못함`);
 }
 s.talents[ti].locked=true;

 const locked=s.talents.filter(x=>x.locked);
 const slots=+QUAL[String(s.zizhi)]?.slots||0;
 if(slots<2)throw Error(`현재 원시 계산 개방 칸은 ${slots}칸입니다. 무한교체/치환은 충돌맥+자맥 2칸이 필요하므로 최소 2칸을 개방하세요.`);
 if(locked.length!==2||!locked.some(x=>+x.id===conflict)||!locked.some(x=>+x.id===target))
   throw Error("기준상태가 '충돌맥+자맥 2칸 잠금'이 아님");

 // 이 prepared가 모든 N행의 동일한 기준상태. 원시 계산에서 선택한 슬롯 수를 그대로 사용한다.
 const prepared=dc(s), basePills=+prepared.total_pills, rows=[];

 let nextPurpleN=null, nextPurpleId=null;
 const effectiveMaxN=slots===2?0:maxN;
 for(let n=0;n<=effectiveMaxN;n++){
   // 중요: 이전 N행에서 이어가지 않는다. 매번 동일 prepared 상태에서 새로 시작.
   let branch=dc(prepared), invalid=false;
   for(let k=0;k<n;k++){
     const beforeIds=new Set(branch.talents.filter(x=>x.locked).map(x=>+x.id));
     const tr=washOnce(branch);
     branch=tr.state; // 2잠금 유지, 원시 계산에서 연 나머지 자유칸 전체를 세수
     const lc=branch.talents.filter(x=>x.locked);
     if(lc.length!==2||!lc.some(x=>+x.id===conflict)||!lc.some(x=>+x.id===target))
       throw Error(`N=${n} 진행 중 잠금쌍 유지 실패`);

     // 남은 자유칸에서 새로운 자맥이 뜨는 순간부터 무한교체 N 창 종료.
     const fresh=branch.talents.find(x=>!x.locked && +BYID.get(+x.id)?.weight===4);
     if(fresh){
       invalid=true;
       nextPurpleId=+fresh.id;
       nextPurpleN=k+1;
       break;
     }
   }
   if(invalid)break;

   const luck=+branch.seed;
   rows.push({
     n,
     pills:+branch.total_pills-basePills,
     luck,
     draws:exchangeSequence(luck,branch,count),
     preparedSeed:+prepared.seed,
     freeSlots:Math.max(0,slots-2),
     nextPurpleN
   });
 }
 rows.nextPurpleN=nextPurpleN;
 rows.nextPurpleId=nextPurpleId;
 rows.openSlots=slots;
 rows.freeSlots=Math.max(0,slots-2);
 return rows;
}
